A few lines to add it

import { attachTap, attachDrag } from 'haptick';

// Any button, tab or card
attachTap(button, { haptic: 'medium' });

// Any slider, carousel or swipe
const haptics = attachDrag(slider);
slider.addEventListener('pointermove', () => {
  if (crossedAStep) haptics.tick();
});
import { HapticTap, useHapticDrag } from 'haptick/react';

<button onClick={buy}>
  Buy <HapticTap haptic="medium" />
</button>

const haptics = useHapticDrag(sliderRef);
<div ref={sliderRef} onPointerMove={() => haptics.tick()} />
